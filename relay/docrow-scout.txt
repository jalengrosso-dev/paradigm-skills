                </div>
              </div>
              {d.file_path && <button onClick={() => openFile(d)} style={{ padding: '6px 12px', borderRadius: 6, fontSize: 11, fontWeight: 500, background: t.bgSubtle, color: t.text, border: '1px solid ' + t.border, cursor: 'pointer' }}>Open</button>}
              {d.file_path && <button onClick={() => triggerVersionUpload(d)} disabled={d.locked} title="Upload new version" style={{ padding: '6px 10px', borderRadius: 6, fontSize: 11, fontWeight: 500, background: 'transparent', color: d.locked ? t.textTer + '60' : t.textSec, border: '1px solid ' + t.border, cursor: d.locked ? 'not-allowed' : 'pointer' }}>+ Version</button>}
              <button onClick={() => { setEditing(d); setModal(true); }} style={{ padding: 6, color: t.textSec, cursor: 'pointer', borderRadius: 4 }}><Edit3 size={14} /></button>
              <button onClick={() => remove(d)} disabled={d.locked} style={{ padding: 6, color: d.locked ? t.textTer + '60' : t.textTer, cursor: d.locked ? 'not-allowed' : 'pointer', borderRadius: 4 }}><Trash2 size={14} /></button>
            </div>
          );
        })}
      </div>
      {modal && <DocModal t={t} clients={clients} opps={opps} initialFile={droppedFile} existing={editing} onClose={() => { setModal(false); setDroppedFile(null); setEditing(null); }} onSaved={() => { setModal(false); setDroppedFile(null); setEditing(null); refresh.docs(); }} />}
    </div>
  );
}
