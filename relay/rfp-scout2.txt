import fs from 'fs';
import path from 'path';

function getApiKey() {
  const direct = process.env.ANTHROPIC_API_KEY;
  if (direct) return direct;
  try {
    const envContent = fs.readFileSync(path.resolve(process.cwd(), '.env'), 'utf-8');
    const match = envContent.match(/ANTHROPIC_API_KEY=(.+)/);
    return match ? match[1].trim() : process.env.ANTHROPIC_API_KEY;
  } catch(e) { return process.env.ANTHROPIC_API_KEY; }
}

// Regex pre-pass: guarantee no clause number is missed even if the AI overlooks it.
// FAR: 52.2xx-xx  DFARS: 252.2xx-xxxx  (also catches alt-agency 2xx.xxx-xxxx forms)
function scanClauses(text) {
  const re = /\b(2?52\.\d{3}-\d{1,4})(?:\s+Alt(?:ernate)?\s+[IVX]+)?/g;
  const found = new Map();
  let m;
  while ((m = re.exec(text)) !== null) {
    const num = m[1];
    if (!found.has(num)) {
      const start = Math.max(0, m.index - 20);
      const end = Math.min(text.length, m.index + 120);
      found.set(num, text.slice(start, end).replace(/\s+/g, ' ').trim());
    }
  }
  return [...found.entries()].map(([number, context]) => ({ number, context }));
}

export default async (req, context) => {
  if (req.method !== 'POST') {
    return new Response(JSON.stringify({ error: 'Method not allowed' }), {
      status: 405, headers: { 'Content-Type': 'application/json' }
    });
  }
  try {
    const { proposal_id, rfp_text, sections } = await req.json();
    if (!proposal_id || !rfp_text) {
      return new Response(JSON.stringify({ error: 'proposal_id and rfp_text required' }), {
        status: 200, headers: { 'Content-Type': 'application/json' }
      });
    }

    const supabaseUrl = process.env.SUPABASE_URL || process.env.VITE_SUPABASE_URL;
    const supabaseKey = process.env.SUPABASE_SERVICE_KEY;

    // Scan the FULL text for clause numbers before truncation — Section I lives at the end.
    const scannedClauses = scanClauses(rfp_text);
    const clauseHint = scannedClauses.length
      ? `\n\nCLAUSE NUMBERS DETECTED IN THE FULL DOCUMENT (you MUST return a row for each one, even if its text is not shown below because it is incorporated by reference):\n${scannedClauses.map(c => `- ${c.number}  [context: ${c.context}]`).join('\n')}`
      : '';

    const sectionList = (sections && sections.length ? sections : ['Executive Summary','Technical Approach','Management Approach','Past Performance','Price','Compliance']).join(', ');

    const prompt = `You are a GovCon compliance analyst. Extract EVERY requirement AND every FAR/DFARS clause from this solicitation.

PART 1 — REQUIREMENTS. Any sentence containing "shall", "must", "will", "is required", "are required", "should" (when directive), or submission instructions (page limits, format, deadlines, volumes).

PART 2 — CLAUSES. Every FAR clause (52.xxx-xx) and DFARS clause (252.xxx-xxxx) referenced anywhere, including Section I clauses incorporated by reference (listed by number with no body text). A clause number with no accompanying text is INCORPORATED BY REFERENCE — still return it.

For each item return:
- requirement_text: for a requirement, the exact requirement quoted or tightly paraphrased. For a clause, a one-line plain-English statement of what the clause obligates (e.g., "Pay Davis-Bacon prevailing wages to laborers and mechanics").
- source_reference: where it came from (e.g., "Section L 3.2", "SOW 4.1", "Section M Factor 2", "Section I").
- requirement_type: one of "shall" (mandatory performance), "submission" (format/instructions), "evaluation" (how they score), "certification" (certs/registrations needed). For a clause, pick the type that best fits its effect — most map to "shall".
- clause_number: the FAR/DFARS number if this item IS a clause (e.g., "52.228-15"), otherwise null.
- clause_title: the official clause title if known (e.g., "Performance and Payment Bonds—Construction"), otherwise null. Only set when clause_number is set.
- by_reference: true if this is a clause listed by number with no body text present (incorporated by reference), false otherwise. Only meaningful when clause_number is set.
- narrative_required: true ONLY if the item requires content in the PROPOSAL (a narrative response, a plan, a citation, or pricing input). Most FAR/DFARS clauses are contract terms the offeror simply complies with and need NO proposal narrative — set false for those. Set true for clauses that drive proposal content, e.g.: Davis-Bacon (wage classifications in pricing), 52.228-15 Performance/Payment Bonds (bonding confirmation), 52.219-9 Small Business Subcontracting Plan (a plan volume), Buy American / 52.225-x (sourcing approach), 52.204-21 / DFARS 252.204-7012 (cybersecurity/CUI approach).
- suggested_section: which proposal section addresses it. Choose from: ${sectionList}. For a comply-silently clause with no narrative, use "Compliance".
- is_hidden: true if a requirement was buried in NARRATIVE prose (SOW/PWS body) rather than an explicit instructions/Section L list. false for items from an obvious instructions/evaluation list or from a clause list.

Respond ONLY with a JSON array. No preamble, no markdown fences. Example:
[{"requirement_text":"Furnish performance and payment bonds each for 100% of contract price.","source_reference":"Section I","requirement_type":"shall","clause_number":"52.228-15","clause_title":"Performance and Payment Bonds—Construction","by_reference":true,"narrative_required":true,"suggested_section":"Price","is_hidden":false}]

SOLICITATION TEXT:
${rfp_text.substring(0, 90000)}${clauseHint}`;

        narrative_required: r.narrative_required === true,
        flow_down: false,
        sort_order: i

    // Enrich clause rows against the authoritative far_clause_reference table:

    try {

        headers: { 'apikey': supabaseKey, 'Authorization': 'Bearer ' + supabaseKey }
      });
      if (refRes.ok) {
        const refRows = await refRes.json();
        const refMap = new Map((Array.isArray(refRows) ? refRows : []).map(x => [String(x.clause_number).trim(), x]));
        for (const row of rows) {
          if (!row.clause_number) continue;
          const ref = refMap.get(row.clause_number);
          if (!ref) continue;
          if (ref.clause_title) row.clause_title = String(ref.clause_title).substring(0, 300);
          row.flow_down = ref.flow_down === true;
        }
      }
    } catch (e) {
      // Reference enrichment is best-effort — never block the insert on it.
    }

    const insertRes = await fetch(supabaseUrl + '/rest/v1/compliance_requirements', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'apikey': supabaseKey,
        'Authorization': 'Bearer ' + supabaseKey,
        'Prefer': 'return=representation'
      },
      body: JSON.stringify(rows)
    });

    const inserted = await insertRes.json();

      count: Array.isArray(inserted) ? inserted.length : 0,
      clauses_detected: scannedClauses.length,
      requirements: inserted

      status: 200, headers: { 'Content-Type': 'application/json' }
    });
  } catch (error) {

      status: 500, headers: { 'Content-Type': 'application/json' }
    });
  }
