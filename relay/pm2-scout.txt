  const QUICK_REQUESTS = ['Sign the consulting agreement', 'Send last 2 years of financials', 'Confirm bonding capacity with your agent', 'Send key personnel resumes', 'Review and approve the proposal draft', 'Complete the onboarding questionnaire'];
  const [requests, setRequests] = useState([]);
  const [title, setTitle] = useState('');
  const [detail, setDetail] = useState('');
  const [busy, setBusy] = useState(false);

  const load = async () => {
    try { setRequests(await db.clientRequests.listByClient(clientId)); } catch(e) {}
  };
  useEffect(() => { load(); }, [clientId]);

  const add = async () => {
    if (!title.trim()) return alert('Request title required');
    setBusy(true);
    try {
      await db.clientRequests.create({ client_id: clientId, title: title.trim(), detail: detail.trim() || null });
      setTitle(''); setDetail('');
      load();
    } catch(e) { alert('Failed: ' + e.message); }
    finally { setBusy(false); }
  };

  const remove = async (r) => {
