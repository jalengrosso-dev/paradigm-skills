        headers: { 'Content-Type': 'application/json' }
      });
    }

    const url = `https://raw.githubusercontent.com/jalengrosso-dev/paradigm-skills/main/skills/${folder}/SKILL.md`;
    const res = await fetch(url);
    
    if (!res.ok) {
      return new Response(JSON.stringify({ error: 'Skill file not found', skill: null }), {
        status: 200,
        headers: { 'Content-Type': 'application/json' }
      });
    }

    const skill = await res.text();

    return new Response(JSON.stringify({ success: true, skill }), {
      status: 200,
      headers: { 'Content-Type': 'application/json' }
    });

  } catch (error) {
    return new Response(JSON.stringify({ error: error.message, skill: null }), {
      status: 500,
      headers: { 'Content-Type': 'application/json' }
    });
  }

function getSystemPrompt(agentId, ctx, memories, skillContent) {
  const agent = AGENTS.find(a => a.id === agentId);
  if (!agent) return 'You are a helpful AI assistant.';
  const base = `You are ${agent.name}, a specialized AI agent inside The Paradigm Universe — a CRM for Paradigm, a government contracting consultancy run by Jay. Your specialty: ${agent.description}. Be direct, operator-grade, and specific. Reference real GovCon mechanics (SAM.gov, FPDS, FAR/DFARS, set-asides, capability statements, CPARS, IDIQ/BPA, etc).`;
  let prompt = ctx ? `${base}\n\n--- CONTEXT ---\n${ctx.summary}\n--- END ---` : base;
  if (skillContent) {
    const trimmedSkill = skillContent.length > 8000 ? skillContent.substring(0, 8000) + '\n[... truncated for context limits]' : skillContent;
    prompt += `\n\n--- SKILL FILE ---\n${trimmedSkill}\n--- END SKILL ---\nFollow these instructions precisely.`;
  }
  if (memories && memories.length > 0) {
    const memoryText = memories.map(m => `[${m.memory_type}] ${m.content}`).join('\n');
    prompt += `\n\n--- BRAIN MEMORY (past intelligence about this entity) ---\n${memoryText}\n--- END MEMORY ---\nUse this memory to inform your response. Reference relevant history when applicable.`;
  }
  return prompt;
}

function getOpeningMessage(agentId, ctx) {
  const agent = AGENTS.find(a => a.id === agentId);
  if (!agent) return "How can I help?";

  if (ctx.type === 'opportunity') { const o = ctx.raw.opportunity; const d = daysUntil(o.due_date); return `Looking at **${o.title}** (${o.solicitation_number}). Stage: ${o.stage}. ${d > 0 ? `${d}d to due.` : 'Past due.'} Bid ${o.bid_no_bid_score}/10. Focus?`; }
  if (ctx.type === 'agency') return `${ctx.raw.agency.abbrev} loaded. ${ctx.raw.agency.touchpoints}/${ctx.raw.agency.touchpoint_goal} touchpoints. Move?`;

}
